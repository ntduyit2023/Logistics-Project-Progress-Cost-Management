\section{Thực nghiệm \& Benchmark}

\begin{frame}{21. Đánh giá Mô hình AI: Tiến trình cải thiện R²}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Phân tích Hiệu suất Dự báo (R²):}
        \begin{itemize}
            \item \textbf{Thách thức dữ liệu:} Với giới hạn tập dữ liệu huấn luyện nhỏ (5 dự án), mô hình khó đạt được khả năng tổng quát hóa hoàn hảo.
            \item \textbf{Hiệu suất đạt được:} Pre-training kết hợp tối ưu tham số (Optuna) giúp mô hình đạt Mean R² = 0.38.
        \end{itemize}
        \vspace{0.2cm}
        \textbf{Vai trò của Mô hình trong Hệ thống:}
        \begin{itemize}
            \item Không nhằm thay thế các thuật toán giải tích truyền thống.
            \item Đóng vai trò cung cấp \textbf{Hệ số nhiễu (Noise factors)} và phân phối xác suất cho mô phỏng Monte Carlo, chuyển bài toán từ học máy thuần túy sang \textbf{tối ưu hóa ngẫu nhiên (stochastic optimization)}.
        \end{itemize}

        \includegraphics[width=\linewidth]{sosanh.jpg}
    \end{columns}
\end{frame}

\begin{frame}{23. Đánh giá Mô hình AI: Mối tương quan theo quy mô đồ thị}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Sự phụ thuộc vào cấu trúc (Graph Structure Dependency):}
        Hiệu năng của mô hình GNN/HGT thể hiện tính tuyến tính đối với quy mô mạng lưới.
        \begin{itemize}
            \item \textbf{C2012-08 (437 tasks):} Đạt hiệu suất mô phỏng tốt (R² = 0.82).
            \item \textbf{C2019-16 (32 tasks):} Mô hình không hội tụ (R² < 0).
        \end{itemize}
        \vspace{0.2cm}
        $\Rightarrow$ \textbf{Đánh giá:} Quy trình huấn luyện hiện tại đáp ứng tính khả thi trong việc hỗ trợ dự báo cho các mạng lưới Logistics quy mô lớn và phức tạp.
