\section{AI Pipeline}

\begin{frame}{16. Kiến trúc Tổng thể (Proposed Architecture)}
    \begin{enumerate}
        \item \textbf{Biểu diễn Dữ liệu (Data Representation):} Chuyển đổi dữ liệu mạng lưới dự án thành cấu trúc Đồ thị Dị thể (Heterogeneous Graph).
        \item \textbf{Học Đặc trưng (Feature Learning):} Ứng dụng mạng nơ-ron đồ thị (HGT) để ngoại suy các tham số nhiễu (noise parameters) bao gồm độ trễ kỳ vọng và phương sai.
        \item \textbf{Mô phỏng Ngẫu nhiên (Stochastic Simulation):} Sử dụng phương pháp Monte Carlo tích hợp phân phối Beta-PERT nhằm định lượng Xác suất Tới hạn (Criticality Index - $CI$).
        \item \textbf{Xây dựng Không gian Trạng thái (State Space Formulation):} Khởi tạo các kịch bản thực thi (Execution Modes) dựa trên phân tích tương quan (Attention Scores) và $CI$.
        \item \textbf{Tối ưu hóa Ràng buộc (Constraint Optimization):} Giải bài toán quy hoạch thỏa mãn ràng buộc (CP-SAT) nhằm xác định tập hợp nghiệm đánh đổi (Trade-off Solutions).
    \end{enumerate}
\end{frame}

\begin{frame}{18. Suy luận và Khai phá Tương quan (Inference \& Attention)}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Cấu trúc Suy luận (Inference Module):}
        \begin{itemize}
            \item \texttt{duration\_factor}: Hàm kích hoạt Softplus kết hợp dịch chuyển $0.5$ nhằm đảm bảo tính dương nghiêm ngặt.
            \item \texttt{expected\_delay}: Hàm kích hoạt ReLU loại trừ nhiễu âm.
            \item \texttt{uncertainty\_sigma}: Ước lượng phương sai thông qua hàm Softplus $+ 0.01$.
        \end{itemize}
        
        \column{0.5\textwidth}
        \textbf{Cơ chế Tập trung Tương tự Cosine (Cosine Attention):}
        Định lượng mức độ phụ thuộc và nút thắt cổ chai (bottleneck) giữa tập Công việc ($T$) và Tài nguyên ($R$).
        \begin{equation}
            \text{sim}(t, r) = \frac{\mathbf{h}_t \cdot \mathbf{h}_r}{\|\mathbf{h}_t\| \|\mathbf{h}_r\|}
        \end{equation}
    \end{columns}

\begin{frame}{18. B4: Mô phỏng Ngẫu nhiên (Stochastic CPM)}
    \textbf{Tích hợp tham số nhiễu ngoại suy vào phân phối Beta-PERT:}
    Tiến trình lấy mẫu ngẫu nhiên (Sampling Process) cho từng công việc qua $N$ vòng lặp:
    \begin{align}
        m &= \text{duration} \times \text{duration\_factor} + \text{expected\_delay} \\
        a &= \max(0.5, \; m \times (1 - 2\sigma)) \\
        b &= m \times (1 + 3\sigma)
    \end{align}
    
    \vspace{0.2cm}
    \textbf{Tổng hợp Phân phối Thống kê Cốt lõi:}
    \begin{itemize}
        \item $P_{50}, P_{80}, P_{95}$: Các mốc thời gian hoàn thành dựa trên độ tin cậy.
        \item \textbf{Criticality Index ($CI_i$):} Tỷ lệ xác suất nút $i$ nằm trên đường găng (Critical Path).
    \end{itemize}
    \textit{Vai trò:} Tham số $CI$ đóng vai trò heuristic định hướng nhằm thu hẹp không gian tìm kiếm (Search Space) trong bước tối ưu tiếp theo.
\end{frame}

\begin{frame}{19. B5: Tối ưu hóa Lịch trình (CP-SAT Optimizer)}
    \textbf{Chiến lược Rút ngắn Tiến độ Dẫn hướng (Heuristic-Guided Crashing):}
    Cơ chế này giới hạn không gian tìm kiếm thông qua việc chỉ áp dụng các phương án nén thời gian (Crashing) lên các nút mạng có $CI$ vượt ngưỡng (threshold), qua đó tối ưu hóa chi phí biên (marginal cost).
    
    \vspace{0.2cm}
    \textbf{Định hình Không gian Giải pháp (Alternative Execution Modes):}
    \begin{table}
        \centering
        \footnotesize
        \begin{tabular}{@{}llcc@{}}
            \toprule
            \textbf{Mode} & \textbf{Chiến lược} & \textbf{Thời gian} & \textbf{Chi phí} \\
            \midrule
            \textbf{0. Normal} & Giữ nguyên thiết kế ban đầu. & 100\% & 100\% \\
            \textbf{1. Overtime} & Khai thác ngoài giờ (CI > 0). & $\sim$66\% & +50\% (Hệ số 1.5x) \\
            \textbf{2. AddRes} & Bổ sung tài nguyên (Dựa trên Attention). & $\sim$70\% & +30\% \\
            \textbf{3. Hybrid} & Phối hợp tối đa hóa nguồn lực (CI $> 90\%$). & $\sim$50\% & +80\% \\
            \bottomrule
        \end{tabular}
    \end{table}
\end{frame}

\begin{frame}{20. B5: Cấu trúc Mô hình Quy hoạch (Constraint Formulation)}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Ràng buộc Lập trình (CP-SAT Constraints):}
        \begin{enumerate}
            \item \textbf{Precedence:} \texttt{AddEndBeforeStart} đảm bảo tính toàn vẹn của quan hệ tiền nhiệm.
            \item \textbf{Resource:} \texttt{AddCumulative} kiểm soát giới hạn tài nguyên khả dụng.
            \item \textbf{Deadline:} Tích hợp hàm phạt (Penalty) vi phạm tiến độ mục tiêu.
            \item \textbf{Multi-Scenario:} Ứng dụng quét lưới (Grid Sweep) tham số chi phí nhằm sinh tập hợp các nghiệm không bị thống trị (Non-dominated Solutions).
        \end{enumerate}
