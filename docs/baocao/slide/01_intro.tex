\section{Mở đầu}

\begin{frame}{1. Bối cảnh \& Vấn đề (Problem Motivation)}
    \textbf{Thực trạng quản lý dự án logistics hiện nay:}
    \begin{itemize}
        \item Chủ yếu sử dụng các công cụ truyền thống (MS Project, Excel) $\rightarrow$ Hạn chế trong khả năng mô phỏng và dự báo rủi ro vượt tiến độ.
        \item \textbf{Số liệu thực tiễn:} 69\% dự án xây dựng bị trễ tiến độ (PMI, 2023 \cite{pmi2023}); dẫn đến trượt chi phí 15–30\%.
    \end{itemize}
    \vspace{0.3cm}
    \textbf{Ba bài toán cốt lõi:}
    \begin{enumerate}
        \item \textbf{Độ phức tạp cấu trúc:} Dự án là mạng lưới các quan hệ phụ thuộc phi tuyến $\rightarrow$ Yêu cầu mô hình hóa dưới dạng Đồ thị.
        \item \textbf{Tính bất định ngẫu nhiên:} Thời gian thực thi tuân theo phân phối xác suất (Beta-PERT), không phải các hằng số định lượng tĩnh.
        \item \textbf{Tối ưu đa mục tiêu:} Nén tiến độ (Crashing) đồng thời tối thiểu hóa tổng chi phí và rủi ro.
    \end{enumerate}
\end{frame}

\begin{frame}{2. Câu hỏi \& Mục tiêu Nghiên cứu}
    \begin{block}{Câu hỏi nghiên cứu}
        \textit{"Làm thế nào để hệ thống hoá và tự động hoá quá trình lập lịch, dự báo rủi ro, và tối ưu hoá đa mục tiêu một dự án logistics phức tạp bằng cách kết hợp Heterogeneous Graph Transformer, mô phỏng Beta-PERT Monte Carlo CPM, và Constraint Programming (CP-SAT)?"}
    \end{block}
    \vspace{0.2cm}
    \textbf{3 Mục tiêu Khoa học:}
    \begin{itemize}
        \item Mô hình hoá dự án dạng \textbf{Heterogeneous Graph} với 3 loại node (Task, Resource, Shift).
        \item Thiết kế pipeline AI: \textbf{HGT Pretraining $\rightarrow$ Inference $\rightarrow$ Beta-PERT MC $\rightarrow$ AI-Guided CP-SAT}.
        \item Hiện thực hệ thống full-stack vận hành thực tế.
    \end{itemize}
\end{frame}
