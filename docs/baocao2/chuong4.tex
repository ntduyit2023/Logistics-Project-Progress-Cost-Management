\chapter{KẾT LUẬN VÀ HƯỚNG PHÁT TRIỂN}

\section{Kết luận chung về đề tài}
\label{sec:4.1_ket_luan_chung}

Đề tài đã hoàn thành xuất sắc việc nghiên cứu lý thuyết, thiết kế kiến trúc và thực nghiệm đánh giá hệ thống Quản trị Tiến độ và Chi phí Dự án Logistics (GLPO). Thông qua việc kết hợp giữa mô hình Học máy Đồ thị Dị thể (Heterogeneous Graph Learning) và giải thuật Tối ưu hóa Quy hoạch Ràng buộc (Constraint Programming), nghiên cứu đã giải quyết trọn vẹn bài toán Lập lịch Dự án Hạn chế Tài nguyên (RCPSP) dưới tác động của rủi ro biến động ngẫu nhiên và cấu trúc chi phí tài chính đa tầng.

\subsection{Đóng góp về mặt lý luận khoa học}
\label{sec:4.1.1_dong_gop_ly_luan}

\begin{enumerate}
    \item \textbf{Mô hình hóa Đồ thị Dị thể:} Chuyển đổi bài toán RCPSP từ biểu diễn phẳng truyền thống sang cấu trúc Đồ thị Dị thể $G = (V, E, \mathcal{T}_v, \mathcal{T}_e)$. Mạng nơ-ron HGT (Heterogeneous Graph Transformer) đã học thành công "nhận thức cấu trúc" (Structural Awareness), đạt $R^2$ trung bình 5-Fold là $\mathbf{0,3734 \pm 0,3744}$, cho phép nhúng trọn vẹn 38 loại chi phí vi mô vào vector biểu diễn của từng công việc.
    \item \textbf{Cơ chế Học tự giám sát (Self-Supervised Learning):} Khắc phục rào cản khan hiếm nhãn dữ liệu thông qua phương pháp Masked Autoencoder. Mô hình tự động trích xuất đặc trưng lan truyền trễ tiến độ mà không phụ thuộc vào nhãn thủ công (Val MAE đạt $0,2343$).
    \item \textbf{Explainable AI (XAI) qua Trọng số Attention:} Trọng số chú ý $\alpha_{ij}^{(h)}$ định vị chính xác $100\%$ các Nút thắt cổ chai găng (Bottlenecks) trên dự án C2012-04. Từ đó, AI dẫn đường cho bộ giải CP-SAT cắt tỉa không gian tìm kiếm, rút ngắn thời gian hội tụ xuống mức mili-giây.
\end{enumerate}

\subsection{Đóng góp về mặt thực tiễn kỹ thuật}
\label{sec:4.1.2_dong_gop_thuc_tien}

\begin{enumerate}
    \item \textbf{Tối ưu Đa mục tiêu (Pareto Frontier):} Dựng thành công đường biên đánh đổi Thời gian - Chi phí. Chiến lược ép tiến độ giúp dự án C2012-04 giảm $21,1\%$ thời gian thi công, hạ rủi ro từ $48,5\%$ xuống $12,0\%$, đồng thời san bằng biểu đồ nhân lực (Giảm hệ số Gini từ $0,316$ xuống $0,100$).
    \item \textbf{Triển khai Vi dịch vụ (Microservices):} Toàn bộ pipeline được đóng gói Docker, tối ưu hóa RAM (chỉ chiếm 3,31 GB cho dự án $10.000$ nút), sẵn sàng triển khai thực tế trên môi trường công nghiệp.
\end{enumerate}

\section{Hạn chế của nghiên cứu}
\label{sec:4.2_han_che_nghien_cuu}

Bên cạnh các thành tựu đạt được, hệ thống GLPO vẫn tồn tại một số hạn chế khách quan cần được phân tích thẳng thắn:
\begin{enumerate}
    \item \textbf{Quy mô và Tính đa dạng của Tập dữ liệu:} Mô hình hiện tại được huấn luyện dựa trên cấu trúc đồ thị của 23 dự án nội bộ. Dù đã sử dụng các kỹ thuật augmentation, tập dữ liệu này vẫn mang tính đặc thù ngành cao. Để chống hiện tượng quá khớp (Overfitting) và áp dụng cho các ngành khác (như IT, Y tế), hệ thống cần một lượng dữ liệu đa dạng hơn rất nhiều.
    \item \textbf{Điểm nghẽn NP-Hard của Bộ giải Tập trung:} Như đã biện luận tại Chương 3, khi số lượng nút công việc $|V| > 10.000$, bộ giải CP-SAT tập trung (Centralized Solver) chắc chắn sẽ vấp phải hiện tượng Bùng nổ tổ hợp (Combinatorial Explosion). AI có thể hướng dẫn thu hẹp không gian, nhưng không thể thay đổi bản chất hàm mũ $O(c^{|V|})$ của bài toán Vận trù học cốt lõi.
    \item \textbf{Thiếu thực nghiệm Phân lập đặc trưng (Ablation Study) tầng sâu:} Đề tài tập trung nhiều vào tính tích hợp toàn hệ thống (End-to-End Pipeline). Tuy nhiên, nghiên cứu chưa có điều kiện thực hiện các thử nghiệm vô hiệu hóa (masking) từng đầu chú ý (Attention Heads) hoặc từng lớp mạng nơ-ron để đo lường định lượng tỷ trọng đóng góp của riêng từng thành phần mạng HGT.
\end{enumerate}

\section{Đề xuất Hướng phát triển}
\label{sec:4.3_huong_phat_trien}

Nhằm khắc phục các hạn chế kỹ thuật và mở rộng quy mô, ba hướng nghiên cứu công nghệ lõi được đề xuất triển khai:

\subsection{Tích hợp Tìm kiếm Lân cận Lớn (LNS) phối hợp CP-SAT}
\label{sec:4.3.1_lns_cpsat}

Để vượt qua rào cản NP-Hard ở siêu dự án ($|V| > 1.000$), nghiên cứu đề xuất sử dụng Tìm kiếm Lân cận Lớn (Large Neighborhood Search - LNS) làm khung điều khiển cho CP-SAT.
Cơ chế lặp sẽ liên tục "Phá hủy" (Destroy) một cụm nhỏ dự án (VD: 15-30\%) và dùng CP-SAT "Tái thiết lập" (Solve) lại cụm đó trong một giới hạn thời gian cực ngắn (100-500ms):
\begin{equation}
S^{(t+1)} = \text{Solve}_{\text{CP-SAT}} \left( \text{Destroy}_{\gamma} \left( S^{(t)} \right) \right)
\end{equation}
Cách tiếp cận này giữ cho không gian tìm kiếm luôn nhỏ gọn, duy trì khả năng tối ưu thời gian thực cho các dự án quy mô siêu lớn.

\subsection{Nâng cấp lên Kiến trúc Graphormer / MARL}
\label{sec:4.3.2_graphormer}

Mô hình HGT hiện tại xử lý thông tin bằng cơ chế truyền tin cục bộ (Local Message Passing). Trên các đường găng kéo dài, tín hiệu có thể bị suy hao (Over-smoothing). Hướng đi tiếp theo là áp dụng \textbf{Graph Transformer (Graphormer)} để tích hợp trực tiếp khoảng cách ngắn nhất $\phi(v_i, v_j)$ vào ma trận Self-Attention, giúp các nút đầu dự án "giao tiếp" ngay lập tức với nút cuối dự án với độ trễ $O(1)$.
Đồng thời, nghiên cứu khả năng dùng Học tăng cường Đa tác tử (MARL), trong đó mỗi thiết bị/nhân sự là một Agent tự đàm phán lịch làm việc, thay vì dùng một bộ giải tập trung.

\subsection{Tích hợp Bản sao số (Digital Twin) và Dữ liệu Thời gian thực}
\label{sec:4.3.3_data_scaling}

Đưa hệ thống GLPO từ một công cụ "Lập lịch tĩnh" trở thành một \textbf{Bản sao số (Digital Twin)} sống động:
\begin{itemize}
    \item \textbf{Luồng dữ liệu IoT \& Telematics:} Kết nối trực tiếp tín hiệu GPS của hạm đội xe, tiến độ từ cảm biến máy xúc và kho bãi WMS.
    \item \textbf{Tái lập lịch động (Dynamic Rescheduling):} Bất cứ khi nào có rủi ro phát sinh (máy hỏng, thời tiết xấu), AI sẽ nhận diện tín hiệu ngay lập tức, tính toán lại độ trễ và kích hoạt CP-SAT tái lập lịch trong vài giây để tự động khắc phục sự cố.
    \item \textbf{Đóng gói SaaS:} Chuẩn hóa API RESTful/gRPC để cắm trực tiếp module AI này vào các hệ thống ERP Enterprise (như SAP, Oracle) hoặc Microsoft Project.
\end{itemize}
